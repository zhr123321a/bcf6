cmake_minimum_required(VERSION 3.16)
project(bcf6_reference LANGUAGES CXX)
add_library(bcdec_ref SHARED native/bcdec_ref.cpp)
set_target_properties(bcdec_ref PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES)
# Keep the ctypes path stable for single- and multi-configuration generators.
set_target_properties(bcdec_ref PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(bcdec_ref PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}")
endforeach()
if(MSVC)
    target_compile_options(bcdec_ref PRIVATE /O2)
else()
    target_compile_options(bcdec_ref PRIVATE -O2 -Wall -Wextra)
endif()
